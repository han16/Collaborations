# ==============================================================================
# FUNCTION: Run Paired T-Tests on Wide Data
# ==============================================================================
# This reusable function handles the distinct layouts of your tabs dynamically.
analyze_tab_paired <- function(data_table, tab_name, num_id_cols) {
  
  #data_table = thesis_data$`Alveolar Bone height`
  #num_id_cols=2
  # 1. Capture the timepoints from row 1 and remove it from the dataset
  timepoints <- as.character(data_table[1, ])
  clean_rows <- data_table[-1, ]
  
  # 2. Fill downward to fix missing labels in the descriptive columns
  for (i in 1:num_id_cols) {
    clean_rows <- clean_rows %>% fill(!!sym(names(clean_rows)[i]), .direction = "down")
  }
  
  # 3. Build a unified Variable Name from the descriptive columns
  clean_rows <- clean_rows %>% 
    unite("Variable", 1:num_id_cols, sep = "_", remove = FALSE)
  
  # 4. Separate T1 and T2 columns using our row 1 map
  t1_indices <- which(timepoints == "T1")
  t2_indices <- which(timepoints == "T2")
  
  # Re-isolate patient IDs for matching
  patient_ids <- names(clean_rows)
  
  # Pivot T1 data to long format
  t1_long <- clean_rows[, c("Variable", patient_ids[t1_indices+1])] %>%
    pivot_longer(cols = -Variable, names_to = "Patient_ID", values_to = "T1_Val") %>%
    mutate(T1_Val = suppressWarnings(as.numeric(T1_Val))) # Cleaned here
  
  # Pivot T2 data to long format
  t2_long <- clean_rows[, c("Variable", patient_ids[t2_indices+1])] %>%
    pivot_longer(cols = -Variable, names_to = "Patient_ID", values_to = "T2_Val") %>%
    mutate(T2_Val = suppressWarnings(as.numeric(T2_Val))) # Cleaned here
  t2_long$Patient_ID=t1_long$Patient_ID
  
  # 5. Merge T1 and T2 by Variable AND Patient_ID to guarantee perfect pairs
  paired_data <- inner_join(t1_long, t2_long, by = c("Variable", "Patient_ID")) %>%
    filter(!is.na(T1_Val) & !is.na(T2_Val)) # Remove incomplete pairs
  
  # 6. Group by each unique site/tooth and run the Paired T-test
  stats_summary <- paired_data %>%
    group_by(Variable) %>%
    summarise(
      Pairs_Count   = n(),
      Mean_T1       = round(mean(T1_Val),4),
      Mean_T2       = round(mean(T2_Val),4),
      Mean_Diff     = round(mean(T2_Val - T1_Val),4),
      # Run t-test safely if there are enough pairs and variance exists
   #   T_Statistic   = if(n() > 1 && var(T2_Val - T1_Val) > 0) t.test(T2_Val, T1_Val, paired = TRUE)$statistic else NA,
      P_Value       = if(n() > 1 && var(T2_Val - T1_Val) > 0) t.test(T2_Val, T1_Val, paired = TRUE)$p.value else NA,
      .groups       = "drop"
    ) %>%
    # --- NEW: MULTIPLE COMPARISON CORRECTIONS ---
    mutate(
      # 'fdr' (or 'BH') controls the False Discovery Rate (Highly recommended for clinical/dental data)
      P_Adj_FDR        = p.adjust(P_Value, method = "fdr"),
      
      # 'bonferroni' is strict and multiplies p-values by the number of tests in this tab
     # P_Adj_Bonferroni = p.adjust(P_Value, method = "bonferroni")
    ) %>%
    # Update significance markers to read off the more reliable FDR adjusted p-value
    mutate(
      Significance_FDR = case_when(
        P_Adj_FDR < 0.001 ~ "***",
        P_Adj_FDR < 0.01  ~ "**",
        P_Adj_FDR < 0.05  ~ "*",
        TRUE              ~ "ns"
      )
    )
    
  
  return(stats_summary)
}

# ==============================================================================
# 1. ORGANIZE AND SPLIT THE DATA

# Fill down the missing 'X' labels (like U4s, U6s) so every row has its group name
icc_cleaned <- icc_data %>%
  fill(X, .direction = "down") %>%
  # Combine Tooth and Landmark to create a completely unique identifier
  mutate(Measurement_Key = paste(X, X.1, sep = "_"))

# Split the data into Time 1 (Rows 1-10) and Time 2 (Rows 11-20)
t1_data <- icc_cleaned[1:10, ]
t2_data <- icc_cleaned[11:20, ]

# Select only the columns representing patients/raters (C001 through X150981)
# Excluding column metadata like X, X.1, X.2
rater_cols <- c("C001", "C002", "C003", "C004", "C005", "C006", "C007", "C008", 
                "C009", "C010", "C011", "C013", "C016", "C017", "X151481", "X151211", "X150981")


# ==============================================================================
# 2. RUN THE ICC LOOP FOR EACH LANDMARK REFERENCE
# ==============================================================================
icc_results <- list()

for (i in 1:10) {
  landmark_name <- t1_data$Measurement_Key[i]
  
  # Extract patient data vectors for T1 and T2, ensuring they are numeric
  v_t1 <- as.numeric(t1_data[i, rater_cols])
  v_t2 <- as.numeric(t2_data[i, rater_cols])
  
  # Bind into a 2-column matrix (Time 1 vs Time 2)
  # Rows = Patients, Columns = Timepoints
  icc_matrix <- cbind(v_t1, v_t2)
  
  # Remove patients that have NA in either T1 or T2 to avoid math failure
  icc_matrix_clean <- na.omit(icc_matrix)
  
  # Run a Two-way, Single-rating, Absolute Agreement ICC Model (ICC(2,1))
  # This matches the standard test-retest guidelines (Koo & Li, 2016)
  icc_test <- icc(
    icc_matrix_clean, 
    model = "twoway", 
    type = "agreement", 
    unit = "single"
  )
  
  # Store core statistics into our list summary dataframe
  icc_results[[landmark_name]] <- tibble(
    Measurement      = landmark_name,
    Valid_Pairs_N    = nrow(icc_matrix_clean),
    ICC_Value        = round(icc_test$value,4),
    Lower_CI_95      = round(icc_test$lbound,4),
    Upper_CI_95      = round(icc_test$ubound,4),
    P_Value          = adjust_pvalue(icc_test$p.value)
  )
}
