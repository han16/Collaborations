# ==============================================================================
# FUNCTION: Run Paired T-Tests on Wide Data
# ==============================================================================
# This reusable function handles the distinct layouts of your tabs dynamically.
analyze_tab_paired <- function(data_table, tab_name, num_id_cols) {
  
  #data_table = thesis_data$`Alveolar Bone height`
  #num_id_cols=2
  # 1. Capture the timepoints from row 1 and remove it from the dataset
  timepoints <- as.character(data_table[1, ])
  clean_rows <- data_table[-1, ]
  
  # 2. Fill downward to fix missing labels in the descriptive columns
  for (i in 1:num_id_cols) {
    clean_rows <- clean_rows %>% fill(!!sym(names(clean_rows)[i]), .direction = "down")
  }
  
  # 3. Build a unified Variable Name from the descriptive columns
  clean_rows <- clean_rows %>% 
    unite("Variable", 1:num_id_cols, sep = "_", remove = FALSE)
  
  # 4. Separate T1 and T2 columns using our row 1 map
  t1_indices <- which(timepoints == "T1")
  t2_indices <- which(timepoints == "T2")
  
  # Re-isolate patient IDs for matching
  patient_ids <- names(clean_rows)
  
  # Pivot T1 data to long format
  t1_long <- clean_rows[, c("Variable", patient_ids[t1_indices+1])] %>%
    pivot_longer(cols = -Variable, names_to = "Patient_ID", values_to = "T1_Val") %>%
    mutate(T1_Val = suppressWarnings(as.numeric(T1_Val))) # Cleaned here
  
  # Pivot T2 data to long format
  t2_long <- clean_rows[, c("Variable", patient_ids[t2_indices+1])] %>%
    pivot_longer(cols = -Variable, names_to = "Patient_ID", values_to = "T2_Val") %>%
    mutate(T2_Val = suppressWarnings(as.numeric(T2_Val))) # Cleaned here
  t2_long$Patient_ID=t1_long$Patient_ID
  
  # 5. Merge T1 and T2 by Variable AND Patient_ID to guarantee perfect pairs
  paired_data <- inner_join(t1_long, t2_long, by = c("Variable", "Patient_ID")) %>%
    filter(!is.na(T1_Val) & !is.na(T2_Val)) # Remove incomplete pairs
  
  # 6. Group by each unique site/tooth and run the Paired T-test
  stats_summary <- paired_data %>%
    group_by(Variable) %>%
    summarise(
      Pairs_Count   = n(),
      Mean_T1       = mean(T1_Val),
      Mean_T2       = mean(T2_Val),
      Mean_Diff     = mean(T2_Val - T1_Val),
      # Run t-test safely if there are enough pairs and variance exists
      T_Statistic   = if(n() > 1 && var(T2_Val - T1_Val) > 0) t.test(T2_Val, T1_Val, paired = TRUE)$statistic else NA,
      P_Value       = if(n() > 1 && var(T2_Val - T1_Val) > 0) t.test(T2_Val, T1_Val, paired = TRUE)$p.value else NA,
      .groups       = "drop"
    ) %>%
    mutate(
      Significance = case_when(
        P_Value < 0.001 ~ "***",
        P_Value < 0.01  ~ "**",
        P_Value < 0.05  ~ "*",
        TRUE            ~ "ns"
      )
    )
  
  return(stats_summary)
}
