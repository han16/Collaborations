# =========================================================================
# PART (2): Within each Sealer, compare Time Points (Repeated Measures ANOVA + Paired)
# =========================================================================
print("\n\n=== PART 2: Comparisons Among Time Points Within Each Sealer ===")
[1] "\n\n=== PART 2: Comparisons Among Time Points Within Each Sealer ==="
# Reshape data to long format and assign a Subject ID per row within each Sealer group
ph_long <- ph_data %>%
  group_by(Sealer) %>%
  mutate(SubjectID = row_number()) %>%
  ungroup() %>%
  pivot_longer(
    cols = all_of(time_cols),
    names_to = "TimePoint",
    values_to = "Score"
  ) %>%
  drop_na(Score)

ph_long$TimePoint <- factor(ph_long$TimePoint, levels = time_cols)
ph_long$Sealer <- as.factor(ph_long$Sealer)

# Loop through each unique sealer to run repeated measures ANOVA and paired comparisons
unique_sealers <- unique(ph_long$Sealer)

for (s in unique_sealers) {
  cat("\n==================================================\n")
  cat("Sealer:", s, "\n")
  cat("==================================================\n")
  
  df_s <- ph_long %>% filter(Sealer == s)
  
  # Repeated Measures ANOVA (SubjectID as the blocking/error term)
  rm_anova <- aov(Score ~ TimePoint + Error(factor(SubjectID) / TimePoint), data = df_s)
  cat("\nRepeated Measures ANOVA:\n")
  summary(rm_anova)
  
  # Fit linear model equivalent for emmeans post-hoc testing
  model_s <- lm(Score ~ TimePoint + factor(SubjectID), data = df_s)
  emms_s <- emmeans(model_s, ~ TimePoint)
  
  # Pairwise paired t-tests across time points with Tukey adjustment
  pairwise_s <- pairs(emms_s, adjust = "tukey")
  
  cat("\nPairwise Comparisons Across Time Points (Paired t-tests, Tukey-adjusted):\n")
  print(pairwise_s)
}

print("\n\n=== PART 2: Comparisons Among Time Points Within Each Group (Repeated Measures) ===")
[1] "\n\n=== PART 2: Comparisons Among Time Points Within Each Group (Repeated Measures) ==="
for (g in groups) {
  cat("\n==================================================\n")
  cat("Group:", g, "\n")
  cat("==================================================\n")
  
  df_g <- viability_data %>% filter(group == g)
  
  # Assign a SubjectID per replicate across time points 
  # (Assumes rows are balanced/ordered identically across each time point)
  df_g <- df_g %>%
    group_by(time) %>%
    mutate(SubjectID = row_number()) %>%
    ungroup()
  
  # 1. One-Way Repeated Measures ANOVA
  rm_anova <- aov(viability ~ time + Error(factor(SubjectID) / time), data = df_g)
  cat("\nOne-Way Repeated Measures ANOVA:\n")
  print(summary(rm_anova))
  
  # 2. Fit linear model equivalent for emmeans (incorporating SubjectID blocking)
  model_g_rm <- lm(viability ~ time + factor(SubjectID), data = df_g)
  emms_g <- emmeans(model_g_rm, ~ time)
  
  # 3. Pairwise comparisons (functions as paired t-tests with Tukey adjustment)
  pairwise_g <- pairs(emms_g, adjust = "tukey")
  
  cat("\nPairwise Paired Comparisons Among Time Points (Tukey-adjusted):\n")
  print(pairwise_g)
}
