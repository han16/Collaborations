library(dplyr)
library(tidyr)
library(emmeans)

# Define the group columns
group_cols <- c("Group.1...ProRoot", "Group.2...Ion.", "Group.3...BC", "Group.4...AH.Plus")

# 1. Add a Row ID for pairing/repeated measures, reshape to long format, and drop NAs
df_long <- endo_sealers %>%
  mutate(RowID = row_number()) %>%
  pivot_longer(
    cols = all_of(group_cols),
    names_to = "Group",
    values_to = "Score"
  ) %>%
  drop_na(Score)

# Clean up group names for readable output
df_long$Group <- gsub("Group\\..*?\\.\\.\\.", "", df_long$Group)
df_long$Group <- sub("\\.$", "", df_long$Group)


# --- (1) ANOVA among ProRoot, Ion, and BC ---
df_3groups <- df_long %>% filter(Group %in% c("ProRoot", "Ion", "BC"))

anova_3 <- aov(Score ~ Group + Error(factor(RowID) / Group), data = df_3groups)
print("--- ANOVA Results (ProRoot, Ion, BC) ---")
